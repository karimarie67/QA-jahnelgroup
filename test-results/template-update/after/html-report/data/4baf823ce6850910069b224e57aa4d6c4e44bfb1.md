(function () {
  'use strict';


  /* ── Logo marquee: duplicate the cells once so the 50% translate
     loops seamlessly. Runs on load — main.js (defer) builds #trust first. ── */
  window.addEventListener('load', function () {
    var track = document.getElementById('trust');
    if (!track || !track.children.length || track.dataset.looped) return;
    track.dataset.looped = '1';
    track.innerHTML += track.innerHTML;
    track.querySelectorAll('img').forEach(function (im) { im.setAttribute('aria-hidden', 'true'); });
  });

  /* ── Capabilities vignette: scroll-driven life-cycle (no loop) ──
     Entry: the plug-in intro plays once in real time (0 → T_POWERED).
     Scroll: typing is scrubbed to the card stack's transit — reversible,
     scrolling up un-types character by character (the steps() keyframes
     scrub per-char for free).
     End: the last stretch past the stack maps to yank + CRT death and
     holds dead there; reversing out of the end zone re-plugs it.
     Exit + re-enter from above: intro replays. Enter from below: starts
     dead (no plug-in), un-plugging in reverse as you scroll up.
     The 5s keyframes stay the source of truth — we pause them and drive
     currentTime. The caret's nested 1s blink is left running live. */
  var capAnim  = document.querySelector('.hr-cap__anim');
  var capStack = document.querySelector('.hr-cap__stack');
  if (capAnim && capStack && !window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
    var T_POWERED = 1200;  // 24%: plugged + powered, typing about to start
    var T_TYPED   = 3300;  // 66%: all four code lines in
    var T_YANK    = 3450;  // 69%: yank begins
    var T_DEAD    = 3950;  // 79%: screen dead, plug back at rest
    var capAnims = [], capCur = 0, capVisible = false, capIntro = false;
    var capRaf = 0, capLastTs = 0;

      capAnims = capAnim.getAnimations({ subtree: true }).filter(function (a) {
        return a.animationName && a.animationName !== 'cx-caret-blink';
      });
      capAnims.forEach(function (a) { a.pause(); });
    };
    var applyCapTime = function (t) {
      for (var i = 0; i < capAnims.length; i++) capAnims[i].currentTime = t;
    };
    // Timeline target from the columns' scroll position.
    // Plug-in: when the section rides up to ~45% of the viewport.
    // Typing: scrubbed from there until the right column's bottom catches
    // the left column (the sticky releases and they rejoin).
    // Unplug: as the rejoined columns scroll away together.
    var capLeft = document.querySelector('.hr-cap__left');
    var capTarget = function () {
      var vh = innerHeight;
      var st = parseFloat(getComputedStyle(capLeft).top) || 120;
      var r = capStack.getBoundingClientRect();
      var startY = vh * 0.45;                    // plug-in cue (columns still together)
      // stack top at the moment the sticky releases (columns rejoin):
      var endTop = st + capLeft.offsetHeight - r.height;
      if (r.top > startY) return 0;              // section not far enough up yet
      var pType = Math.min(1, Math.max(0, (startY - r.top) / Math.max(1, startY - endTop)));
      var pYank = Math.min(1, Math.max(0, (endTop - r.top) / (vh * 0.3)));
      var t = T_POWERED + pType * (T_TYPED - T_POWERED);
      if (pYank > 0) t = Math.max(t, T_YANK + pYank * (T_DEAD - T_YANK));
      return t;
    };
    var capPlugged = false; // has the intro fired for the current pin crossing?
    grabCapAnims();          // kill the free-running CSS loop right away
    applyCapTime(0);         // idle unplugged until the columns separate
    var capFrame = function (ts) {
      capRaf = 0;
      if (!capVisible) return;
      var dt = capLastTs ? Math.min(0.1, (ts - capLastTs) / 1000) : 0.016;
      capLastTs = ts;
      var tgt = capTarget();
      if (capIntro) {
        // Intro plays at 2.5x — snappy plug-in (~0.5s), independent of scroll
        capCur = Math.min(T_POWERED, capCur + dt * 2500);
        if (capCur >= T_POWERED) capIntro = false;
        applyCapTime(capCur);
      } else if (!capPlugged) {
        // Waiting, unplugged, for the columns to separate
        if (tgt > 0) { capPlugged = true; capIntro = true; }
      } else {
        if (Math.abs(tgt - capCur) > 0.5) {
          capCur += (tgt - capCur) * (1 - Math.exp(-12 * dt));
          if (Math.abs(tgt - capCur) < 1) capCur = tgt;
          applyCapTime(capCur);
        } // settled: no style writes
        // Fully reversed back above the pin: re-arm the intro
        if (tgt === 0 && capCur < 1) { capPlugged = false; capCur = 0; }
      }
      capRaf = requestAnimationFrame(capFrame);
    };
    new IntersectionObserver(function (entries) {
      entries.forEach(function (en) {
        capVisible = en.isIntersecting;
        if (!capVisible) return;
        grabCapAnims();
        var tgt = capTarget();
        if (tgt >= T_YANK) {                   // entering from below: already dead
          capCur = T_DEAD; capIntro = false; capPlugged = true;
        } else if (tgt > 0) {                  // entering mid-stack: play the plug-in
          capCur = 0; capIntro = true; capPlugged = true;
        } else {                               // above the pin: idle, unplugged
          capCur = 0; capIntro = false; capPlugged = false;
        }
        applyCapTime(capCur);
        capLastTs = 0;
        if (!capRaf) capRaf = requestAnimationFrame(capFrame);
      });
    }, { threshold: 0 }).observe(capLeft.closest('section'));
  }

  /* ── Ring stat count-up: numbers tick from 0 to target once the
     awards section scrolls into view. The \"+\" is a separate static
     span and never animates. Reduced-motion: land on the final value. ── */
  var ringCounts = document.querySelectorAll('.hr-ring__count');
  if (ringCounts.length) {
    var reduceMo = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    var fmt = function (n) { return n.toLocaleString('en-US'); };
    if (reduceMo) {
      ringCounts.forEach(function (el) { el.textContent = fmt(+el.dataset.count); });
    } else {
      ringCounts.forEach(function (el) { el.textContent = '0'; });
      var runCount = function (el) {

        var step = function (ts) {
          if (!start) start = ts;
          var p = Math.min(1, (ts - start) / dur);
          var eased = 1 - Math.pow(1 - p, 3);          // easeOutCubic
          el.textContent = fmt(Math.round(eased * target));
          if (p < 1) requestAnimationFrame(step);
        };
        requestAnimationFrame(step);
      };

      new IntersectionObserver(function (entries, io) {
        entries.forEach(function (en) {
          if (!en.isIntersecting) return;
          io.disconnect();
          ringCounts.forEach(runCount);
        });
      }, { threshold: 0.35 }).observe(ringSection);
    }
  }

})();
